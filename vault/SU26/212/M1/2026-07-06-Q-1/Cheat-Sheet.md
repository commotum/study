
$$
\omega  =\frac{d \theta}{d t} \qquad \alpha =\frac{d \omega}{d t}\qquad  \theta  =\int \omega d t \qquad  \omega  =\int \alpha d t
$$
$$
s = r\theta \qquad v = r\omega \qquad a_t = r\alpha
$$
$$
\omega_f = \omega_0 + \alpha t \qquad \theta_f = \theta_0 + \omega_0 t + \frac{1}{2}\alpha t^2 \qquad \omega_f^2 = \omega_0^2 + 2\alpha \Delta\theta
$$
$$
x \leftrightarrow \theta, \quad v \leftrightarrow \omega, \quad a \leftrightarrow \alpha
$$
$$
a_r=\frac{v^2}{r}=r \omega^2 \qquad a_t=\frac{d v}{d t}=r \alpha \qquad |\vec{a}|=\sqrt{a_r^2+a_t^2}
$$
$$
f=\frac{1}{T} \qquad \omega=2\pi f=\frac{2\pi}{T} \qquad v=\frac{2\pi r}{T}
$$
$$
\sum F_r = m a_r = m\frac{v^2}{r} = m\omega^2 r \qquad \sum F_t = m a_t
$$
$$
f_{s,\max}=\mu_s N \qquad \mu_{s,\text{flat}}=\frac{v^2}{rg} \qquad \omega_{\max,\text{turntable}}=\sqrt{\frac{\mu_s g}{r}}
$$
$$
v_{\text{banked}}=\sqrt{rg\tan\theta} \qquad v_{\max,\text{banked}}=\sqrt{rg\frac{\sin\theta+\mu_s\cos\theta}{\cos\theta-\mu_s\sin\theta}}
$$
$$
N_{\mathrm{top}}=m\left(g-\frac{v^2}{r}\right)=m\left(g-\omega^2 r\right)
$$
$$
N_{\mathrm{bottom}}=m\left(g+\frac{v^2}{r}\right)=m\left(g+\omega^2 r\right)
$$
$$
T\cos\theta=mg \qquad T\sin\theta=m\frac{v^2}{r}
$$
$$
a_r=\frac{T_{\text{tens}}}{m}+g\cos\theta \qquad a_t=g\sin\theta \qquad |\vec{a}|=\sqrt{a_r^2+a_t^2}
$$
$$
\Delta y_{\max}=\frac{v_{0y}^2}{2g}=\frac{v_0^2\sin^2\theta}{2g} \qquad v_{0y}=v_0\sin\theta
